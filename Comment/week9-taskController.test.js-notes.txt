require("dotenv").config(); //to get the database URLs from .env
process.env.DATABASE_URL = process.env.TEST_DATABASE_URL; //point to the test database
const prisma = require("../db/prisma");
const httpMocks = require("node-mocks-http");
const { EventEmitter } = require("events");
const waitForRouteHandlerCompletion = require("./waitForRouteHandlerCompletion");

const {
  create,
  index,
  show,
  update,
  deleteTask,
} = require("../controllers/taskController");

//a few useful globals
let user1 = null;
let user2 = null;
let saveRes = null;
let saveData = null;
let saveTaskId = null;

//beforeAll  will empty the database and create the user records needed for the tests
beforeAll(async () => {
  //clear database
  await prisma.Task.deleteMany(); //delete all tasks
  await prisma.User.deleteMany(); //delete all users

  user1 = await prisma.User.create({
    data: {
      name: "Bob",
      email: "bob@sample.com",
      hashedPassword: "nonsense",
    },
  });
  user2 = await prisma.User.create({
    data: {
      name: "Alice",
      email: "alice@sample.com",
      hashedPassword: "nonsense",
    },
  });
});
afterAll(() => {
  prisma.$disconnect(); //to terminate cleanly
});

describe("testing task creation", () => {
  it("14. cant create a task without a user id", async () => {
    const req = httpMocks.createRequest({
      //create a mock (fake/simulated) request object
      method: "POST",
      body: { title: "first task:" },
    });
    saveRes = httpMocks.createResponse({
      // create a mock (fake/simulated) response object
      eventEmitter: EventEmitter, //to listen for res.on("finish", ...), it needs an eventEmitter
    });
    expect.assertions(1); //run expect once"
    try {
      await waitForRouteHandlerCompletion(create, req, saveRes);
    } catch (err) {
      expect(err.name).toBe("TypeError"); //err.name is the name of the error.,want to test if the name of the error is "TypeError"
    }
  });
  it("15. can't create a task with a bogus user id", async () => {
    //bogus means fake
    const req = httpMocks.createRequest({
      method: "POST",
      body: { title: "first task" },
    });
    req.user = { id: -1 }; // ★ a user id that does not exist in the DB
    saveRes = httpMocks.createResponse({
      eventEmitter: EventEmitter,
    });
    expect.assertions(1);
    try {
      await waitForRouteHandlerCompletion(create, req, saveRes);
    } catch (err) {
      expect(err.name).toBe("PrismaClientKnownRequestError");
    }
  });
  it("16. If you have a valid user id, create() succeeds (saveRes.statusCode should be 201)", async () => {
    const req = httpMocks.createRequest({
      method: "POST",
      body: { title: "first task" },
    });
    req.user = { id: user1.id };
    saveRes = httpMocks.createResponse({
      //empty response
      eventEmitter: EventEmitter,
    });
    expect.assertions(1);
    await waitForRouteHandlerCompletion(create, req, saveRes); //saveRes acts as res
    // calls create(req, saveRes, next) internally:
    //   - create reads data FROM req (req.body, req.user)
    //   - create WRITES its result INTO saveRes (via res.status()/res.json())
    expect(saveRes.statusCode).toBe(201);
  });
  it("17. The object returned from the create() call has the expected title", async () => {
    saveData = saveRes._getJSONData(); //saveRes from 16
    expect(saveData.title).toBe("first task");
  });
  it("18. The object has the right value for isCompleted", async () => {
    saveData = saveRes._getJSONData(); //saveRes from 16
    expect(saveData.isCompleted).toBe(false);
  });
  it("19. The object does not have any value for userId", async () => {
    saveData = saveRes._getJSONData(); //saveRes acts as res //saveRes from 16
    //ex, if create returns res.status(201).json({ id: 3, title: "first task", isCompleted: false });
    //then call saveRes._getJSONData() and you'll get back saveData = { id: 3, title: "first task", isCompleted: false }
    expect(saveData.userId).toBeUndefined();
    saveTaskId = saveData.id;
  });
});

describe("testing the update and delete of tasks", () => {
  it("28.  User1 can set the task corresponding to saveTaskId to isCompleted: true", async () => {
    const req = httpMocks.createRequest({
      method: "PATCH",
      body: { isCompleted: true }, //title and priority is not required, so no need to provide(it already exists in the database when created)
    });
    req.user = { id: user1.id }; //make sure the user is user1
    req.params = { id: saveTaskId.toString() }; //make sure the task is the one created
    saveRes = httpMocks.createResponse({
      //response write from request and get in saveData //// create an empty mock response; update() will write its result into this, which we later read via saveData
      eventEmitter: EventEmitter,
    });
    await waitForRouteHandlerCompletion(update, req, saveRes);
    saveData = saveRes._getJSONData();
    expect(saveData.isCompleted).toBe(true);
  });
  it("29. User2 can't do this", async () => {
    const req = httpMocks.createRequest({
      method: "PATCH",
      body: { isCompleted: true },
    });
    req.user = { id: user2.id };
    req.params = { id: saveTaskId.toString() };
    saveRes = httpMocks.createResponse({
      eventEmitter: EventEmitter,
    });
    await waitForRouteHandlerCompletion(update, req, saveRes);
    expect(saveRes.statusCode).toBe(404);
  });
  it("30. User2 can't delete this task", async () => {
    const req = httpMocks.createRequest({
      method: "DELETE",
    });
    req.user = { id: user2.id };
    req.params = { id: saveTaskId.toString() };
    saveRes = httpMocks.createResponse({
      eventEmitter: EventEmitter,
    });
    await waitForRouteHandlerCompletion(deleteTask, req, saveRes);
    saveData = saveRes._getJSONData();
    expect(saveRes.statusCode).toBe(404);
  });
  it("31. User1 can delete this task", async () => {
    const req = httpMocks.createRequest({
      method: "DELETE",
    });
    req.user = { id: user1.id };
    req.params = { id: saveTaskId.toString() };
    saveRes = httpMocks.createResponse({
      eventEmitter: EventEmitter,
    });
    await waitForRouteHandlerCompletion(deleteTask, req, saveRes);
    expect(saveRes.statusCode).toBe(200);
  });
  it("32. Retrieving user1's tasks now returns a 404", async () => {
    const req = httpMocks.createRequest({
      method: "GET",
    });
    req.user = { id: user1.id };
    req.params = { id: saveTaskId.toString() };
    saveRes = httpMocks.createResponse({
      eventEmitter: EventEmitter,
    });
    await waitForRouteHandlerCompletion(show, req, saveRes);
    expect(saveRes.statusCode).toBe(404);
  });
});
